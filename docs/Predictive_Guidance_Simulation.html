 65def target_location(t, target_states):
 66    """
 67    Compute the target aircraft's position at time t, given the
 68    history of target_states (for determining the start of the curve).
 69    Always assume starting in the +X direction, then turning in the YZ plane at the specified angle.
 70
 71    0 <= t <= Straight_time: straight line
 72
 73    Straight_time < t <= Straight_time + curve_time: curved turn
 74
 75    Straight_time + curve_time < t <= Straight_time + curve_time + Straight_time2: straight line after turn
 76
 77    t > Straight_time + curve_time + Straight_time2: hold last position
 78
 79    returns a numpy array of shape (3,) representing the true (x, y, z) position of the target aircraft at time t
 80    This is the data generation function, not the noisy measurements, which are generated later.
 81    """
 82
 83    global curve_start_x, curve_start_y, curve_start_z
 84    global curve_initialized
 85
 86    global straight2_start_x, straight2_start_y, straight2_start_z
 87    global straight2_initialized
 88
 89    global center_x, center_y, center_z
 90
 91    if 0 <= t <= Straight_time:
 92
 93        x = aircraft_start_loc[0] + targ_vel * t
 94        y = aircraft_start_loc[1]
 95        z = aircraft_start_loc[2]
 96
 97    elif Straight_time < t <= Straight_time + curve_time:
 98
 99        tc = t - Straight_time
100
101        if not curve_initialized:
102
103            if len(target_states) > 0:
104                curve_start_x = target_states[-1, 0]
105                curve_start_y = target_states[-1, 1]
106                curve_start_z = target_states[-1, 2]
107
108            else:
109                curve_start_x = (
110                    aircraft_start_loc[0]
111                    + targ_vel * Straight_time
112                )
113                curve_start_y = aircraft_start_loc[1]
114                curve_start_z = aircraft_start_loc[2]
115
116            center_x = curve_start_x
117            center_y = (
118                curve_start_y
119                + radius * np.cos(yz_angle)
120            )
121            center_z = (
122                curve_start_z
123                + radius * np.sin(yz_angle)
124            )
125
126            curve_initialized = True
127
128        angle = tc * turn_angle / curve_time
129
130        arc_angle = -np.pi / 2 + angle
131
132        x = center_x + radius * np.cos(arc_angle)
133
134        y = (
135            center_y
136            + radius
137            * np.sin(arc_angle)
138            * np.cos(yz_angle)
139        )
140
141        z = (
142            center_z
143            + radius
144            * np.sin(arc_angle)
145            * np.sin(yz_angle)
146        )
147
148    elif (
149        Straight_time + curve_time
150        < t
151        <= Straight_time + curve_time + Straight_time2
152    ):
153
154        if not straight2_initialized:
155
156            straight2_start_x = target_states[-1, 0]
157            straight2_start_y = target_states[-1, 1]
158            straight2_start_z = target_states[-1, 2]
159
160            straight2_initialized = True
161
162        ts = t - (Straight_time + curve_time)
163
164        dx = np.cos(turn_angle)
165        dy = np.sin(turn_angle) * np.cos(yz_angle)
166        dz = np.sin(turn_angle) * np.sin(yz_angle)
167
168        x = straight2_start_x + targ_vel * ts * dx
169        y = straight2_start_y + targ_vel * ts * dy
170        z = straight2_start_z + targ_vel * ts * dz
171
172    else:
173
174        if straight2_initialized:
175
176            ts_max = Straight_time2
177
178            dx = np.cos(turn_angle)
179            dy = np.sin(turn_angle) * np.cos(yz_angle)
180            dz = np.sin(turn_angle) * np.sin(yz_angle)
181
182            x = (
183                straight2_start_x
184                + targ_vel * ts_max * dx
185            )
186
187            y = (
188                straight2_start_y
189                + targ_vel * ts_max * dy
190            )
191
192            z = (
193                straight2_start_z
194                + targ_vel * ts_max * dz
195            )
196
197        else:
198
199            x = (
200                aircraft_start_loc[0]
201                + targ_vel * Straight_time
202            )
203
204            y = aircraft_start_loc[1]
205            z = aircraft_start_loc[2]
206
207    return np.array([x, y, z])

386def point_segment_distance(p, a, b):
387    """
388    input: p = True point(x,y,z) of target, a = segment start, b = segment end
389
390    Compute the shortest distance from the missile to the path between 2 missile locations, a and b,
391    i.e. how close the missile came to the aircraft during this time step
392    and return the closest point on the segment to p.
393
394    Assume detonation is based on true proximity, not the noisy measurements, so this is used to determine if the missile hit the aircraft.
395    """
396    ab = b - a
397    denom = np.dot(ab, ab)
398    if denom < 1e-12:
399        return np.linalg.norm(p - a), a.copy()
400    u = np.clip(np.dot(p - a, ab) / denom, 0.0, 1.0)
401    closest = a + u * ab
402    return np.linalg.norm(p - closest), closest

506            final_noisy_pt
507        )

512            t_noisy,

514            final_noisy_pt[1],
515            final_noisy_pt[2]

537            noisy_pts,

630            # the missile will be at its current position + missile_speed * tau * heading_unit_vector (direction of the missile)

632            #square both sides to get a quadratic equation in tau, then solve for tau

651                    if roots:

656            else:

659

661            # so launch happens as soon as a velocity estimate exists)
662            if missile_state["last_t"] is None:
663                step_dt = 0.0
664            else:
665                step_dt = t_now - missile_state["last_t"]
666
667            missile_state["last_t"] = t_now
668
669            heading = aim - m_old #vector from missile to aim point, which is the direction the missile should move in this time step
670            heading_norm = np.linalg.norm(heading)
671
672            if heading_norm > 1e-9: #prevent division by zero, if the missile is already at the aim point, it does not move
673                m_new = m_old + heading / heading_norm * missile_speed * step_dt #missile steps toward aim point at constant speed, distance scaled by the time step
674            else:
675                m_new = m_old
676
677            # Hit check against the TRUE aircraft position, along this step
678            miss, closest = point_segment_distance(
679                target_states[frame], m_old, m_new
680            )
681
682            missile_state["min_dist"] = min(missile_state["min_dist"], miss)
683
684            if miss <= hit_radius: #if its within the hit radius, the missile has hit the target
685                missile_state["hit"] = True
686                missile_state["hit_time"] = t_now
687                m_new = closest
688
689            missile_state["pos"] = m_new #update the missile position to the new position after this time step
690            missile_state["trail"].append(m_new.copy()) #update the missile trail with the new position it has moved to
691
692            trail = np.array(missile_state["trail"])
693
694            missile_point.set_data([m_new[0]], [m_new[1]])
695            missile_point.set_3d_properties([m_new[2]])
696
697            missile_trail.set_data(trail[:, 0], trail[:, 1])
698            missile_trail.set_3d_properties(trail[:, 2])
699
700            if missile_state["hit"]:
701                aim_point.set_data([], [])
702                aim_point.set_3d_properties([])
703            else:
704                aim_point.set_data([aim[0]], [aim[1]])
705                aim_point.set_3d_properties([aim[2]])
706

388    """
389    input: p = True point(x,y,z) of target, a = segment start, b = segment end
390
391    Compute the shortest distance from the missile to the path between 2 missile locations, a and b,
392    i.e. how close the missile came to the aircraft during this time step
393    and return the closest point on the segment to p.
394
395    Assume detonation is based on true proximity, not the noisy measurements, so this is used to determine if the missile hit the aircraft.
396    """
397    ab = b - a
398    denom = np.dot(ab, ab)
399    if denom < 1e-12:
400        return np.linalg.norm(p - a), a.copy()
401    u = np.clip(np.dot(p - a, ab) / denom, 0.0, 1.0)
402    closest = a + u * ab
403    return np.linalg.norm(p - closest), closest
